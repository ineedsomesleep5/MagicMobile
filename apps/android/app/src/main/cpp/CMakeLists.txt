cmake_minimum_required(VERSION 3.22)
project(magicmobile_android C)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../..")
set(ENGINE "${ROOT}/packages/ondevice-engine")
set(ARTIFACT "${ROOT}/apps/android/native-artifact")
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Only the real Android ARM64 engine is supported")
endif()
if(NOT EXISTS "${ARTIFACT}/libmmengine.so")
  message(FATAL_ERROR "Missing verified full XMage Android library. No mock fallback.")
endif()
add_library(mmengine SHARED IMPORTED)
set_target_properties(mmengine PROPERTIES IMPORTED_LOCATION "${ARTIFACT}/libmmengine.so")
add_library(magicmobile_jni SHARED mm_android_jni.c "${ENGINE}/native/mm_graal_backend.c" "${ENGINE}/swift/Sources/CMagicEngine/mm_runtime.c")
target_include_directories(magicmobile_jni PRIVATE "${ENGINE}/native" "${ENGINE}/swift/Sources/CMagicEngine/include" "${ARTIFACT}/include")
target_compile_options(magicmobile_jni PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_options(magicmobile_jni PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,defs")
target_link_libraries(magicmobile_jni PRIVATE mmengine log)
